## Concurrent Idempotency

Command:

```bash
python3 - <<'PY'
import asyncio
import json
import time
import httpx

payload = {
    "model": "llama3.2:1b",
    "provider": "ollama",
    "messages": [
        {
            "role": "user",
            "content": "Reply with exactly: v6-concurrent-idem-ok",
        }
    ],
}
key = f"v6-concurrent-{int(time.time())}"

async def main():
    async with httpx.AsyncClient(timeout=90) as client:
        async def one(i):
            response = await client.post(
                "http://localhost:8000/v1/chat/completions",
                headers={"Idempotency-Key": key},
                json=payload,
            )
            body = response.json()
            return {
                "i": i,
                "status": response.status_code,
                "retry_after": response.headers.get("retry-after"),
                "id": body.get("id"),
                "detail": body.get("detail"),
                "content": (
                    ((body.get("choices") or [{}])[0].get("message") or {})
                    .get("content")
                ),
            }

        results = await asyncio.gather(*(one(i) for i in range(10)))

    print(json.dumps({
        "key": key,
        "results": results,
        "status_counts": {
            str(status): sum(1 for item in results if item["status"] == status)
            for status in sorted({item["status"] for item in results})
        },
        "retry_after_values": sorted({
            item["retry_after"] for item in results if item["retry_after"]
        }),
        "unique_success_ids": sorted({
            item["id"] for item in results if item["id"]
        }),
    }, indent=2))

asyncio.run(main())
PY
```
